CREATE DATABASE IF NOT EXISTS mental_health_db
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE mental_health_db;

CREATE TABLE IF NOT EXISTS core_datasetmetadata (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    filename VARCHAR(255) NOT NULL,
    upload_date DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    file_path VARCHAR(500) NOT NULL,
    rows INT DEFAULT 0,
    `columns` INT DEFAULT 0,
    FOREIGN KEY (user_id) REFERENCES auth_user(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS core_predictionhistory (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    prediction_date DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    gender VARCHAR(10) NOT NULL,
    financial_issues VARCHAR(5) NOT NULL,
    family_issues VARCHAR(100) NOT NULL,
    study_hours FLOAT NOT NULL,
    health_issues VARCHAR(100) NOT NULL,
    friends_issues VARCHAR(100) NOT NULL,
    time_with_friends FLOAT NOT NULL,
    working_hours VARCHAR(5) NOT NULL,
    unpleasant_events VARCHAR(5) NOT NULL,
    academic_stress VARCHAR(5) NOT NULL,
    career_stress VARCHAR(5) NOT NULL,
    criticism VARCHAR(5) NOT NULL,
    conflicts VARCHAR(5) NOT NULL,
    predicted_stress VARCHAR(20) NOT NULL,
    risk_score FLOAT DEFAULT 0.0,
    model_used VARCHAR(50) DEFAULT 'Random Forest',
    FOREIGN KEY (user_id) REFERENCES auth_user(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS core_chathistory (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    user_message TEXT NOT NULL,
    bot_response TEXT NOT NULL,
    timestamp DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES auth_user(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE INDEX idx_prediction_user_date ON core_predictionhistory(user_id, prediction_date);
CREATE INDEX idx_dataset_user ON core_datasetmetadata(user_id);
CREATE INDEX idx_chat_user ON core_chathistory(user_id);

SELECT 'Database setup complete!' AS status;
